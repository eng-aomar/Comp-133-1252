#include <stdio.h>

int main() {
    int r=2, c=3;
    int matrix[r][c];

    printf("Enter 6 integers:\n");

    for (int i = 0; i < r; i++) {
        for (int j = 0; j < c; j++) {
            printf("matrix[%d][%d]=", i,j);
            scanf("%d", &matrix[i][j]);
        }
    }

    printf("\nMatrix:\n");

    for (int i = 0; i < 2; i++) {
        for (int j = 0; j < 3; j++) {
            printf("%d ", matrix[i][j]);
        }
        printf("\n");
    }

    return 0;
}
