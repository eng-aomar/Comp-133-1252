#include <stdio.h>

int main() {
    int matrix[2][3] = {
        {1, 2, 3},
        {4, 5, 6}
    };
   printf("\tQ1\tQ2\tQ3\n");
    for (int i = 0; i < 2; i++) {
            printf("S[%d]\t", i+1);
        for (int j = 0; j < 3; j++) {
            printf("%d\t",matrix[i][j]);
        }
        printf("\n");
    }

    return 0;
}
